# 一.sort类

## (一)、sort + cmp

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
int a[100100];

bool cmp(int x, int y) {
    return x > y;
}

int main() {
    cin >> n;
    for(int i = 1; i <= n; i ++) {
        cin >> a[i];
    }
    sort(a + 1, a + n + 1, cmp);
    for(int i = 1; i <= n; i ++) {
        cout << a[i] << " ";
    }
}
```

## (二)、sort + swap

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
int a[100100];

int main() {
    cin >> n;
    for(int i = 1; i <= n; i ++) {
        cin >> a[i];
    }
    sort(a + 1, a + n + 1);
    for(int i = 1; i <= n / 2; i ++) {
        swap(a[i], a[n - i + 1]);
    }
    for(int i = 1; i <= n; i ++) {
        cout << a[i] << " ";
    }
}
```

# 二.二叉排序树

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
struct TreeNode {
    int l, r, v;
} a[100010];

inline void dfs(int p) {
    if(a[p].r)
        dfs(a[p].r);
    cout << a[p].v << " ";
    if(a[p].l)
        dfs(a[p].l);
}

int main() {
    cin >> n;
    for(int i = 1; i <= n; i ++) {
        cin >> a[i].v;
    }
    for(int i = 2; i <= n; i ++) {
        int p = 1;
        while(true) {
            if(a[i].v <= a[p].v) {
                if(a[p].l) {
                    p = a[p].l;
                } else {
                    a[p].l = i;
                    break;
                }
            } else {
                if(a[p].r) {
                    p = a[p].r;
                } else {
                    a[p].r = i;
                    break;
                }
            }
        }
    }
    dfs(1);
}
```
